# --- COMMANDS (Keypad Press -> MQTT Publish) ---
# Button 1: Toggle the main kitchen light
CMD  : 1 : TOGGLE      : home/kitchen/light/set

# Button 2: Turn on the "All Lights" scene
CMD  : 2 : SCENE_ON    : home/groups/all/set

# Button 3: Volume Up for media player
CMD  : 3 : VOL_UP      : home/livingroom/audio/set

# --- STATUS (MQTT Message -> Update ActuatorState[ID]) ---
# Map kitchen light status to ActuatorState index 0
STAT : 0 : KITCHEN_LT : home/kitchen/light/status

# Map bedroom light status to ActuatorState index 1
STAT : 1 : BEDROOM_LT : home/bedroom/light/status

# Map garage door status to ActuatorState index 2
STAT : 2 : GARAGE_DR  : home/garage/door/status
